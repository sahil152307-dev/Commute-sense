    46→    46→// ---------- Helpers ----------
    47→    47→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    48→    48→  const people: Person[] = [];
    49→    49→  const clothingColors = [
    50→    50→    '#e91e63', '#5d4037', '#1b5e20', '#0d47a1', '#4a148c',
    51→    51→    '#bf360c', '#37474f', '#212121', '#f9a825', '#00695c',
    52→    52→  ];
    53→    53→
    54→    54→  for (let i = 0; i < count; i++) {
    55→    55→    const zoneIndex = i % ZONE_DEFS.length;
    56→    56→    const zd = ZONE_DEFS[zoneIndex];
    57→    57→    const bw = 10 + Math.random() * 6;
    58→    58→    const bh = 20 + Math.random() * 10;
    59→    59→    people.push({
    60→    60→      x: zd.xMin * canvasW + Math.random() * (zd.xMax - zd.xMin) * canvasW,
    61→    61→      y: zd.yMin * canvasH + Math.random() * (zd.yMax - zd.yMin) * canvasH,
    62→    62→      confidence: 0.78 + Math.random() * 0.21,
    63→    63→      id: i + 1,
    64→    64→      vx: (Math.random() - 0.5) * 0.015,
    65→    65→      vy: (Math.random() - 0.5) * 0.01,
    66→    66→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    67→    67→      bodyH: bh,
    68→    68→      bodyW: bw,
    69→    69→      zoneIndex,
    70→    70→    });
    71→    71→  }
    72→    72→  return people;
    73→    73→}
