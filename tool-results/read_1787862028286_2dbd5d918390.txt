   104→function formatTime(iso: string): string {
   105→  try {
   106→    const d = new Date(iso);
   107→    return d.toLocaleTimeString('en-US', { hour12: false, hour: '2-digit', minute: '2-digit', second: '2-digit' });
   108→  } catch {
   109→    return iso;
   110→  }
   111→}
   112→
   113→function formatTimeShort(iso: string): string {
   114→  try {
   115→    const d = new Date(iso);
   116→    const now = new Date();
   117→    const diffMs = now.getTime() - d.getTime();
   118→    const diffMin = Math.floor(diffMs / 60000);
   119→    if (diffMin < 1) return 'Just now';
   120→    if (diffMin < 60) return `${diffMin}m ago`;
   121→    return `${Math.floor(diffMin / 60)}h ago`;
   122→  } catch {
   123→    return iso;
   124→  }
   125→}

   314→// ---- Main Component ----
   315→export function FleetConsole() {
   316→  const [routeStatuses, setRouteStatuses] = useState<RouteStatus[]>([]);
   317→  const [alerts, setAlerts] = useState<FleetAlert[]>([]);
   318→  const [idleVehicleIds, setIdleVehicleIds] = useState<string[]>([]);
   319→  const [selectedVehicle, setSelectedVehicle] = useState<string>('');
   320→  const [selectedRoute, setSelectedRoute] = useState<string>('');
   321→  const [dispatching, setDispatching] = useState(false);
   322→  const [loading, setLoading] = useState(true);
   323→  const [soundMuted, setSoundMuted] = useState(false);
   324→  const [emergencies, setEmergencies] = useState<EmergencyEvent[]>(initialEmergencies);
   325→  const [selectedEmergency, setSelectedEmergency] = useState<EmergencyEvent | null>(null);
   326→  const [emergencyVehicle, setEmergencyVehicle] = useState<string>('');
   327→  const [emergencyDispatching, setEmergencyDispatching] = useState(false);
   328→  const playedAlertIds = useRef<Set<string>>(new Set());
   329→  const [showRouteMap, setShowRouteMap] = useState(false);
   330→  const [dispatchedEmergencies, setDispatchedEmergencies] = useState<Set<string>>(new Set());
   331→
   332→  // Overcrowded routes (demand > 85)
   333→  const overcrowdedRoutes = useMemo(
   334→    () => routeStatuses.filter((r) => r.utilizationPct > 85),
   335→    [routeStatuses]
   336→  );
   337→
   338→  // Idle vehicles from mock data (rich objects)
   339→  const idleVehicles: Vehicle[] = useMemo(() => getIdleVehicles(), []);
   340→
   341→  // Unresolved emergencies
   342→  const activeEmergencies = useMemo(
   343→    () => emergencies.filter(e => !e.resolved),
   344→    [emergencies]
   345→  );
   346→
   347→  // Toggle sound
   348→  const toggleSound = useCallback(() => {
   349→    const next = !soundMuted;
   350→    setSoundMuted(next);
   351→    setMuted(next);
   352→  }, [soundMuted]);
   353→
   354→  // Fetch fleet status
   355→  const fetchStatus = useCallback(async () => {
   356→    try {
   357→      const res = await fetch('/api/v1/fleet/status');
   358→      if (!res.ok) return;
   359→      const data: FleetStatusResponse = await res.json();
   360→      if (data.success) {
   361→        setRouteStatuses(data.routes);
   362→        setAlerts((prev) => {
   363→          const newAlerts = data.alerts.filter(
   364→            (a) => !prev.some((p) => p.timestamp === a.timestamp && p.message === p.message)
   365→          );
   366→          return [...newAlerts, ...prev].slice(0, 50);
   367→        });
   368→        setIdleVehicleIds(data.idleVehicles);
   369→      }
   370→    } catch {
   371→      // silently fail
   372→    } finally {
   373→      setLoading(false);
   374→    }
   375→  }, []);
   376→
   377→  useEffect(() => {
   378→    fetchStatus();
   379→    const interval = setInterval(fetchStatus, 8000);
   380→    return () => clearInterval(interval);
   381→  }, [fetchStatus]);
   382→
   383→  // Simulate new emergencies every 20-40s
   384→  useEffect(() => {
   385→    const scheduleNext = () => {
   386→      const delay = 20000 + Math.random() * 20000; // 20-40s
   387→      return setTimeout(() => {
   388→        const newEvent = generateRandomEmergency();
   389→        setEmergencies(prev => [newEvent, ...prev].slice(0, 20));
   390→        // Play alert sound
   391→        if (!playedAlertIds.current.has(newEvent.id)) {
   392→          playedAlertIds.current.add(newEvent.id);
   393→          playAlertSound(newEvent.type);
   394→        }
   395→        // Also add to alert feed
   396→        setAlerts(prev => [{
   397→          type: newEvent.type,
   398→          vehicleId: newEvent.vehicleId,
   399→          routeName: newEvent.routeName,
   400→          message: newEvent.message,
   401→          timestamp: newEvent.timestamp,
   402→        }, ...prev].slice(0, 50));
   403→        scheduleNext();
   404→      }, delay);
   405→    };
   406→    const timer = scheduleNext();
   407→    return () => clearTimeout(timer);
   408→  }, []);
   409→
   410→  // Play sounds for initial emergencies on first load
   411→  useEffect(() => {
   412→    const timer = setTimeout(() => {
   413→      for (const emg of initialEmergencies) {
   414→        if (!playedAlertIds.current.has(emg.id)) {
   415→          playedAlertIds.current.add(emg.id);
   416→          playAlertSound(emg.type);
   417→        }
   418→      }
   419→    }, 500);
   420→    return () => clearTimeout(timer);
   421→  }, []);
   422→
   423→  // Handle re-route dispatch
   424→  const handleDispatch = async () => {
   425→    if (!selectedVehicle || !selectedRoute) {
   426→      toast.error('Select a vehicle and a target route.');
   427→      return;
   428→    }
   429→    setDispatching(true);
   430→    try {
   431→      const res = await fetch('/api/v1/fleet/re-route', {
   432→        method: 'POST',
   433→        headers: { 'Content-Type': 'application/json' },
   434→        body: JSON.stringify({ sourceRouteId: '', targetRouteId: selectedRoute, vehicleId: selectedVehicle }),
   435→      });
   436→      const data = await res.json();
   437→      if (data.success) {
   438→        toast.success(data.message, { description: data.updatedCapacityRelief });
   439→        setSelectedVehicle('');
   440→        setSelectedRoute('');
   441→        await fetchStatus();
   442→      } else {
   443→        toast.error(data.error || 'Dispatch failed.');
   444→      }
   445→    } catch {
   446→      toast.error('Network error. Please try again.');
   447→    } finally {
   448→      setDispatching(false);
   449→    }
   450→  };
   451→
   452→  // Handle emergency dispatch
   453→  const handleEmergencyDispatch = async () => {
   454→    if (!emergencyVehicle || !selectedEmergency) {
   455→      toast.error('Select a vehicle to dispatch.');
   456→      return;
   457→    }
   458→    setEmergencyDispatching(true);
   459→    try {
   460→      const res = await fetch('/api/v1/fleet/emergency-dispatch', {
   461→        method: 'POST',
   462→        headers: { 'Content-Type': 'application/json' },
   463→        body: JSON.stringify({ vehicleId: emergencyVehicle, emergencyStopId: selectedEmergency.stopId }),
   464→      });
   465→      const data = await res.json();
   466→      if (data.success) {
   467→        toast.success(data.message, {
   468→          description: `Routed via safest path. ETA: ${data.routeData.safeTime} min. Avoiding: ${data.routeData.avoidedZones.join(', ') || 'None'}`,
   469→        });
   470→        setEmergencyVehicle('');
   471→        setSelectedEmergency(null);
   472→        setShowRouteMap(false);
   473→        // Mark emergency as resolved
   474→        setEmergencies(prev => prev.map(e =>
   475→          e.id === selectedEmergency.id ? { ...e, resolved: true } : e
   476→        ));
   477→        setDispatchedEmergencies(prev => new Set([...prev, selectedEmergency.id]));
   478→        playAlertSound('INFO');
   479→      } else {
   480→        toast.error(data.error || 'Emergency dispatch failed.');
   481→      }
   482→    } catch {
   483→      toast.error('Network error. Please try again.');
   484→    } finally {
   485→      setEmergencyDispatching(false);
   486→    }
   487→  };
   488→
   489→  const handleSelectEmergency = (emg: EmergencyEvent) => {
   490→    setSelectedEmergency(emg);
   491→    setShowRouteMap(true);
   492→    playAlertSound(emg.type);
   493→  };
   494→
   495→  const handleResolveEmergency = (emgId: string) => {
   496→    setEmergencies(prev => prev.map(e => e.id === emgId ? { ...e, resolved: true } : e));
   497→    setDispatchedEmergencies(prev => new Set([...prev, emgId]));
   498→    toast.success('Emergency marked as resolved.');
   499→  };
   500→
   501→  const selectedEmergencyVehicle = emergencyVehicle ? getVehicle(emergencyVehicle) : undefined;
   502→
   503→  return (
   504→    <div className="flex flex-col gap-4 w-full">
   505→      {/* ---- 0. Emergency Alert Banner (top-most, full width) ---- */}
   506→      <AnimatePresence>
   507→        {activeEmergencies.length > 0 && (
   508→          <motion.div
   509→            initial={{ opacity: 0, height: 0 }}
   510→            animate={{ opacity: 1, height: 'auto' }}
   511→            className="space-y-2"
   512→          >
   513→            <div className="flex items-center justify-between rounded-lg border border-red-500/50 bg-red-500/10 px-4 py-2.5">
   514→              <div className="flex items-center gap-3">
   515→                <motion.div
   516→                  animate={{ scale: [1, 1.2, 1] }}
   517→                  transition={{ duration: 0.8, repeat: Infinity }}
   518→                >
   519→                  <AlertTriangle className="size-5 text-red-400" />
   520→                </motion.div>
   521→                <div>
   522→                  <p className="text-sm font-bold text-red-400">
   523→                    {activeEmergencies.length} Active Emergenc{activeEmergencies.length === 1 ? 'y' : 'ies'}
   524→                  </p>
   525→                  <p className="text-xs text-red-300/70">
   526→                    {activeEmergencies.reduce((s, e) => s + e.passengersStranded, 0)} passengers need assistance
   527→                  </p>
   528→                </div>
   529→              </div>
   530→              <button
   531→                onClick={toggleSound}
