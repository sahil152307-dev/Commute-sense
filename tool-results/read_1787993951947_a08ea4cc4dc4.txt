    36→// ---------- Zone definitions (proportional) ----------
    37→// Layout: Sky/buildings (top) → Road (middle) → Sidewalk/Platform (bottom)

    39→const ZONE_DEFS = [
    40→  // Zone A: near shelter entrance (platform) — boarding queue area
    41→  { xMin: 0.42, xMax: 0.56, yMin: 0.77, yMax: 0.91 },
    42→  // Zone B: inside bus shelter area — people waiting under shelter
    43→  { xMin: 0.56, xMax: 0.90, yMin: 0.77, yMax: 0.93 },
    44→  // Zone C: platform area left of shelter — people walking on platform
    45→  { xMin: 0.15, xMax: 0.38, yMin: 0.78, yMax: 0.91 },
    46→  // Zone D: far left platform edge
    47→  { xMin: 0.03, xMax: 0.14, yMin: 0.79, yMax: 0.91 },

    50→// ---------- Helpers ----------
    51→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    52→  const people: Person[] = [];
    53→  const clothingColors = [
    54→    '#e91e63', '#5d4037', '#1b5e20', '#0d47a1', '#4a148c',
    55→    '#bf360c', '#37474f', '#212121', '#f9a825', '#00695c',
    56→  ];
    57→
    58→  for (let i = 0; i < count; i++) {
    59→    const zoneIndex = i % ZONE_DEFS.length;
    60→    const zd = ZONE_DEFS[zoneIndex];
    61→    const bw = 10 + Math.random() * 6;
    62→    const bh = 20 + Math.random() * 10;
    63→    people.push({
    64→      x: zd.xMin * canvasW + Math.random() * (zd.xMax - zd.xMin) * canvasW,
    65→      y: zd.yMin * canvasH + Math.random() * (zd.yMax - zd.yMin) * canvasH,
    66→      confidence: 0.78 + Math.random() * 0.21,
    67→      id: i + 1,
    68→      vx: (Math.random() - 0.5) * 0.015,
    69→      vy: (Math.random() - 0.5) * 0.01,
    70→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    71→      bodyH: bh,
    72→      bodyW: bw,
    73→      zoneIndex,
    74→      boarding: false,
    75→      boarded: false,
    76→    });
    77→  }
    78→  return people;
    79→}

    98→function generateHistory(current: number, points = 30) {
    99→  const hist: number[] = [];
   100→  for (let i = 0; i < points - 1; i++) {
   101→    hist.push(Math.max(0, current + Math.floor(Math.random() * 8) - 4));
   102→  }
   103→  hist.push(current);
   104→  return hist;
   105→}
