    49→    49→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    50→    50→  const people: Person[] = [];
    51→    51→  const clothingColors = [
    52→    52→    '#e91e63', '#5d4037', '#1b5e20', '#0d47a1', '#4a148c',
    53→    53→    '#bf360c', '#37474f', '#212121', '#f9a825', '#00695c',
    54→    54→  ];
    55→    55→
    56→    56→  for (let i = 0; i < count; i++) {
    57→    57→    const zoneIndex = i % ZONE_DEFS.length;
    58→    58→    const zd = ZONE_DEFS[zoneIndex];
    59→    59→    const bw = 10 + Math.random() * 6;
    60→    60→    const bh = 20 + Math.random() * 10;
    61→    61→    people.push({
    62→    62→      x: zd.xMin * canvasW + Math.random() * (zd.xMax - zd.xMin) * canvasW,
    63→    63→      y: zd.yMin * canvasH + Math.random() * (zd.yMax - zd.yMin) * canvasH,
    64→    64→      confidence: 0.78 + Math.random() * 0.21,
    65→    65→      id: i + 1,
    66→    66→      vx: (Math.random() - 0.5) * 0.015,
    67→    67→      vy: (Math.random() - 0.5) * 0.01,
    68→    68→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    69→    69→      bodyH: bh,
    70→    70→      bodyW: bw,
    71→    71→      zoneIndex,
    72→    72→      boarding: false,
    73→    73→      boarded: false,
    74→    74→    });
    75→    75→  }
    76→    76→  return people;
    77→    77→}
