    39→// ---------- Helpers ----------
    40→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    41→  const people: Person[] = [];
    42→  const clothingColors = [
    43→    '#e91e63', // pink/magenta
    44→    '#5d4037', // dark brown
    45→    '#1b5e20', // dark olive
    46→    '#0d47a1', // navy
    47→    '#4a148c', // deep purple
    48→    '#bf360c', // dark orange
    49→    '#37474f', // blue grey
    50→    '#212121', // near black
    51→    '#f9a825', // amber
    52→    '#00695c', // teal dark
    53→  ];
    54→
    55→  // Define zones where people's FEET can appear (body extends upward from y)
    56→  const zones = [
    57→    // Zone A: near bus shelter entrance (sidewalk)
    58→    { xMin: canvasW * 0.55, xMax: canvasW * 0.65, yMin: canvasH * 0.70, yMax: canvasH * 0.78 },
    59→    // Zone B: inside bus shelter
    60→    { xMin: canvasW * 0.62, xMax: canvasW * 0.88, yMin: canvasH * 0.40, yMax: canvasH * 0.72 },
    61→    // Zone C: platform/far sidewalk (below road)
    62→    { xMin: canvasW * 0.15, xMax: canvasW * 0.45, yMin: canvasH * 0.76, yMax: canvasH * 0.90 },
    63→    // Zone D: near bus front, on sidewalk
    64→    { xMin: canvasW * 0.02, xMax: canvasW * 0.12, yMin: canvasH * 0.70, yMax: canvasH * 0.80 },
    65→  ];
    66→
    67→  for (let i = 0; i < count; i++) {
    68→    const zone = zones[i % zones.length];
    69→    const bw = 10 + Math.random() * 6;
    70→    const bh = 20 + Math.random() * 10;
    71→    people.push({
    72→      x: zone.xMin + Math.random() * (zone.xMax - zone.xMin),
    73→      y: zone.yMin + Math.random() * (zone.yMax - zone.yMin),
    74→      confidence: 0.78 + Math.random() * 0.21,
    75→      id: i + 1,
    76→      vx: (Math.random() - 0.5) * 0.02,
    77→      vy: (Math.random() - 0.5) * 0.015,
    78→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    79→      bodyH: bh,
    80→      bodyW: bw,
    81→    });
    82→  }
    83→  return people;
    84→}

   103→function generateHistory(current: number, points = 30) {
   104→  const hist: number[] = [];
   105→  for (let i = 0; i < points - 1; i++) {
   106→    hist.push(Math.max(0, current + Math.floor(Math.random() * 8) - 4));
   107→  }
   108→  hist.push(current);
   109→  return hist;
