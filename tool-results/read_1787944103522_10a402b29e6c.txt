    36→// ---------- Zone definitions (proportional) ----------
    37→const ZONE_DEFS = [
    38→  // Zone A: near bus shelter entrance (sidewalk) — right of center
    39→  { xMin: 0.55, xMax: 0.65, yMin: 0.76, yMax: 0.88 },
    40→  // Zone B: inside bus shelter area — people waiting

    42→  // Zone C: platform/far sidewalk (well below road)

    48→// ---------- Helpers ----------
    49→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    50→  const people: Person[] = [];
    51→  const clothingColors = [
    52→    '#e91e63', '#5d4037', '#1b5e20', '#0d47a1', '#4a148c',
    53→    '#bf360c', '#37474f', '#212121', '#f9a825', '#00695c',
    54→  ];
    55→
    56→  for (let i = 0; i < count; i++) {
    57→    const zoneIndex = i % ZONE_DEFS.length;
    58→    const zd = ZONE_DEFS[zoneIndex];
    59→    const bw = 10 + Math.random() * 6;
    60→    const bh = 20 + Math.random() * 10;
    61→    people.push({
    62→      x: zd.xMin * canvasW + Math.random() * (zd.xMax - zd.xMin) * canvasW,
    63→      y: zd.yMin * canvasH + Math.random() * (zd.yMax - zd.yMin) * canvasH,
    64→      confidence: 0.78 + Math.random() * 0.21,
    65→      id: i + 1,
    66→      vx: (Math.random() - 0.5) * 0.015,
    67→      vy: (Math.random() - 0.5) * 0.01,
    68→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    69→      bodyH: bh,
    70→      bodyW: bw,
    71→      zoneIndex,
    72→      boarding: false,
    73→      boarded: false,
    74→    });
    75→  }
    76→  return people;
    77→}

    96→function generateHistory(current: number, points = 30) {
    97→  const hist: number[] = [];
    98→  for (let i = 0; i < points - 1; i++) {
    99→    hist.push(Math.max(0, current + Math.floor(Math.random() * 8) - 4));
   100→  }
   101→  hist.push(current);
   102→  return hist;
   103→}
