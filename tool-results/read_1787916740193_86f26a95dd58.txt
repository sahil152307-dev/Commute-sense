    39→// ---------- Helpers ----------
    40→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    41→  const people: Person[] = [];
    42→  const clothingColors = [
    43→    '#e91e63', // pink/magenta
    44→    '#5d4037', // dark brown
    45→    '#1b5e20', // dark olive
    46→    '#0d47a1', // navy
    47→    '#4a148c', // deep purple
    48→    '#bf360c', // dark orange
    49→    '#37474f', // blue grey
    50→    '#212121', // near black
    51→    '#f9a825', // amber
    52→    '#00695c', // teal dark
    53→  ];
    54→
    55→  // Define zones where people can appear
    56→  const zones = [
    57→    // On the road / near bus
    58→    { xMin: 40, xMax: canvasW * 0.45, yMin: canvasH * 0.45, yMax: canvasH * 0.75 },
    59→    // Near bus shelter
    60→    { xMin: canvasW * 0.55, xMax: canvasW * 0.9, yMin: canvasH * 0.35, yMax: canvasH * 0.7 },
    61→    // Platform / sidewalk
    62→    { xMin: 40, xMax: canvasW * 0.9, yMin: canvasH * 0.72, yMax: canvasH * 0.88 },
    63→  ];
    64→
    65→  for (let i = 0; i < count; i++) {
    66→    const zone = zones[i % zones.length];
    67→    const bw = 10 + Math.random() * 6;
    68→    const bh = 20 + Math.random() * 10;
    69→    people.push({
    70→      x: zone.xMin + Math.random() * (zone.xMax - zone.xMin),
    71→      y: zone.yMin + Math.random() * (zone.yMax - zone.yMin),
    72→      confidence: 0.78 + Math.random() * 0.21,
    73→      id: i + 1,
    74→      vx: (Math.random() - 0.5) * 0.25,
    75→      vy: (Math.random() - 0.5) * 0.15,
    76→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    77→      bodyH: bh,
    78→      bodyW: bw,
    79→    });
    80→  }
    81→  return people;
    82→}

   101→function generateHistory(current: number, points = 30) {
   102→  const hist: number[] = [];
   103→  for (let i = 0; i < points - 1; i++) {
   104→    hist.push(Math.max(0, current + Math.floor(Math.random() * 8) - 4));
   105→  }
   106→  hist.push(current);
   107→  return hist;
   108→}
