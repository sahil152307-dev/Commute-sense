    57→function describeArc(cx: number, cy: number, r: number, startAngle: number, endAngle: number) {
    58→  const start = polarToCartesian(cx, cy, r, endAngle);
    59→  const end = polarToCartesian(cx, cy, r, startAngle);
    60→  const largeArcFlag = endAngle - startAngle <= 180 ? '0' : '1';
    61→  return ['M', start.x, start.y, 'A', r, r, 0, largeArcFlag, 0, end.x, end.y].join(' ');
    62→}
    63→
    64→function polarToCartesian(cx: number, cy: number, r: number, angleDeg: number) {
    65→  const rad = ((angleDeg - 90) * Math.PI) / 180;
    66→  return { x: cx + r * Math.cos(rad), y: cy + r * Math.sin(rad) };
    67→}

   304→// ─── Speed History Sparkline ─────────────────────────────────────────────────
   305→
   306→function SpeedChart({ currentSpeed }: { currentSpeed: number }) {
   307→  const points = useMemo(() => {
   308→    const data: number[] = [];
   309→    for (let i = 0; i < 20; i++) {
   310→      const variation = (Math.random() - 0.5) * 16;
   311→      data.push(Math.max(0, Math.min(80, currentSpeed + variation)));
   312→    }
   313→    // Ensure last point is close to current
   314→    data[19] = currentSpeed;
   315→    return data;
   316→  }, [currentSpeed]);
   317→
   318→  const width = 320;
   319→  const height = 80;
   320→  const padding = 4;
   321→  const maxVal = Math.max(...points, 1);
   322→
   323→  const xStep = (width - padding * 2) / (points.length - 1);
   324→
   325→  const coords = points.map((p, i) => ({
   326→    x: padding + i * xStep,
   327→    y: height - padding - (p / maxVal) * (height - padding * 2),
   328→  }));
   329→
   330→  // Smooth curve using cardinal spline-like approach (quadratic bezier through midpoints)
   331→  let smoothPath = `M ${coords[0].x} ${coords[0].y}`;
   332→  for (let i = 0; i < coords.length - 1; i++) {
   333→    const cp1x = (coords[i].x + coords[i + 1].x) / 2;
   334→    const cp1y = coords[i].y;
   335→    const cp2x = (coords[i].x + coords[i + 1].x) / 2;
   336→    const cp2y = coords[i + 1].y;
   337→    smoothPath += ` C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${coords[i + 1].x} ${coords[i + 1].y}`;
   338→  }
   339→
   340→  const areaPath =
   341→    smoothPath + ` L ${coords[coords.length - 1].x} ${height} L ${coords[0].x} ${height} Z`;
   342→
   343→  const lineColor = speedColor(currentSpeed);
   344→
   345→  return (
   346→    <svg width="100%" height={height} viewBox={`0 0 ${width} ${height}`} preserveAspectRatio="none">
   347→      <defs>
   348→        <linearGradient id="speedGrad" x1="0" y1="0" x2="0" y2="1">
   349→          <stop offset="0%" stopColor={lineColor} stopOpacity="0.3" />
   350→          <stop offset="100%" stopColor={lineColor} stopOpacity="0" />
   351→        </linearGradient>
   352→      </defs>
   353→      <motion.path
   354→        d={areaPath}
   355→        fill="url(#speedGrad)"
   356→        initial={{ opacity: 0 }}
   357→        animate={{ opacity: 1 }}
   358→        transition={{ duration: 0.5 }}
   359→      />
   360→      <motion.path
   361→        d={smoothPath}
   362→        fill="none"
   363→        stroke={lineColor}
   364→        strokeWidth="2"
   365→        strokeLinecap="round"
   366→        initial={{ pathLength: 0 }}
   367→        animate={{ pathLength: 1 }}
   368→        transition={{ duration: 1, ease: 'easeOut' }}
   369→      />
   370→      {/* End dot */}
   371→      <motion.circle
   372→        cx={coords[coords.length - 1].x}
   373→        cy={coords[coords.length - 1].y}
   374→        r="3"
   375→        fill={lineColor}
   376→        initial={{ scale: 0 }}
   377→        animate={{ scale: 1 }}
   378→        transition={{ delay: 0.8, duration: 0.3 }}
   379→      />
   380→    </svg>
   381→  );
   382→}

   562→// ─── Trip Duration Calculator ────────────────────────────────────────────────
   563→
   564→function calcTripDuration(timestamp: string): string {
   565→  const now = Date.now();
   566→  const then = new Date(timestamp).getTime();
   567→  const diffMs = Math.abs(now - then);
   568→  const hours = Math.floor(diffMs / 3600000);
   569→  const minutes = Math.floor((diffMs % 3600000) / 60000);
   570→  if (hours > 0) return `${hours}h ${minutes}m`;
   571→  return `${minutes}m`;
   572→}

   579→  const [currentTime, setCurrentTime] = useState(Date.now());
   580→
   581→  const selectedDriver = useMemo(
   582→    () => data.find((d) => d.vehicleId === selectedVehicleId),
   583→    [data, selectedVehicleId]
   584→  );
   585→
   586→  const isDrowsy = useMemo(
   587→    () =>
   588→      !!selectedDriver &&
   589→      (selectedDriver.drowsinessFlag || selectedDriver.earRatio < 0.2),
   590→    [selectedDriver]
   591→  );
   592→
   593→  // Auto-select first driver on mount
   594→  useEffect(() => {
   595→    if (data.length > 0 && !selectedVehicleId) {
   596→      setSelectedVehicleId(data[0].vehicleId);
   597→    }
   598→  }, [data, selectedVehicleId]);
   599→
   600→  // Poll API every 3 seconds
   601→  useEffect(() => {
   602→    const interval = setInterval(async () => {
   603→      try {
   604→        const res = await fetch(`/api/v1/telematics?vehicleId=${selectedVehicleId}`);
   605→        if (res.ok) {
   606→          const json = await res.json();
   607→          if (json?.data) {
   608→            // Update the specific driver data in state
   609→            setData((prev) =>
   610→              prev.map((d) => (d.vehicleId === selectedVehicleId ? { ...d, ...json.data } : d))
   611→            );
   612→          }
   613→        }
   614→      } catch {
   615→        // Fallback: simulate slight random variations locally
   616→        setData((prev) =>
   617→          prev.map((d) => {
   618→            if (d.vehicleId !== selectedVehicleId) return d;
   619→            return {
   620→              ...d,
   621→              speedKmph: Math.max(0, Math.min(80, d.speedKmph + (Math.random() - 0.5) * 6)),
   622→              earRatio: Math.max(0.05, Math.min(0.45, d.earRatio + (Math.random() - 0.5) * 0.03)),
   623→              stabilityScore: Math.max(20, Math.min(100, d.stabilityScore + (Math.random() - 0.5) * 3)),
   624→              timestamp: new Date().toISOString(),
   625→            };
   626→          })
   627→        );
   628→      }
   629→      setCurrentTime(Date.now());
   630→    }, 3000);
   631→
   632→    return () => clearInterval(interval);
   633→  }, [selectedVehicleId]);
   634→
   635→  // Re-key for speed chart on speed change
   636→  const speedChartKey = selectedDriver ? Math.round(selectedDriver.speedKmph / 5) : 0;
   637→
   638→  return (
   639→    <ScrollArea className="h-full">
   640→      <div className="p-4 lg:p-6 space-y-5">
   641→        {/* Header + Selector */}
   642→        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   643→          <div>
   644→            <h2 className="text-lg font-semibold text-foreground flex items-center gap-2">
   645→              <ShieldAlert className="size-5" style={{ color: TEAL }} />
   646→              Driver Telematics & Drowsiness Alert
   647→            </h2>
   648→            <p className="text-xs text-muted-foreground mt-0.5">
   649→              Edge-AI real-time driver monitoring system
   650→            </p>
   651→          </div>
   652→          <Select value={selectedVehicleId} onValueChange={setSelectedVehicleId}>
   653→            <SelectTrigger className="w-full sm:w-[280px]">
   654→              <SelectValue placeholder="Select driver..." />
   655→            </SelectTrigger>
   656→            <SelectContent>
   657→              {data.map((d) => (
   658→                <SelectItem key={d.vehicleId} value={d.vehicleId}>
   659→                  <span className="flex items-center gap-2">
   660→                    <span
   661→                      className="size-2 rounded-full"
   662→                      style={{
   663→                        backgroundColor: d.drowsinessFlag || d.earRatio < 0.2 ? RED : GREEN,
   664→                      }}
   665→                    />
   666→                    {d.driverName}
   667→                    <span className="text-muted-foreground text-xs">({d.vehicleId})</span>
   668→                  </span>
   669→                </SelectItem>
   670→              ))}
   671→            </SelectContent>
   672→          </Select>
   673→        </div>
   674→
   675→        {/* Drowsiness Alert Banner */}
   676→        <AnimatePresence>
   677→          {isDrowsy && selectedDriver && (
   678→            <motion.div
   679→              initial={{ opacity: 0, height: 0 }}
   680→              animate={{ opacity: 1, height: 'auto' }}
   681→              exit={{ opacity: 0, height: 0 }}
   682→              transition={{ duration: 0.3 }}
   683→            >
   684→              <div className="animate-pulse-alert rounded-lg border-2 border-red-500/60 bg-red-600 px-5 py-4 text-white shadow-lg shadow-red-500/20">
   685→                <div className="flex items-start gap-3">
   686→                  <AlertTriangle className="size-7 mt-0.5 shrink-0" />
   687→                  <div className="flex-1 min-w-0">
   688→                    <div className="text-lg font-bold tracking-wide">
   689→                      🚨 DROWSINESS ALERT
   690→                    </div>
   691→                    <div className="text-sm mt-1 opacity-95">
   692→                      Driver <span className="font-semibold">{selectedDriver.driverName}</span> ({selectedDriver.vehicleId}) is showing signs of drowsiness.
   693→                    </div>
   694→                    <div className="flex flex-wrap items-center gap-3 mt-2 text-xs opacity-80">
   695→                      <span>
   696→                        EAR Ratio: <span className="font-bold text-white">{selectedDriver.earRatio.toFixed(2)}</span>
   697→                      </span>
   698→                      <span className="text-red-200">•</span>
   699→                      <span>
   700→                        Stability: <span className="font-bold text-white">{selectedDriver.stabilityScore}%</span>
   701→                      </span>
   702→                      <span className="text-red-200">•</span>
   703→                      <span>
   704→                        Speed: <span className="font-bold text-white">{Math.round(selectedDriver.speedKmph)} km/h</span>
   705→                      </span>
   706→                    </div>
   707→                    <div className="mt-2 text-xs font-medium bg-red-500/30 rounded px-2.5 py-1 inline-block">
   708→                      ⚠️ Immediate intervention recommended — Alert dispatched to fleet supervisor
   709→                    </div>
   710→                  </div>
   711→                </div>
   712→              </div>
   713→            </motion.div>
   714→          )}
   715→        </AnimatePresence>
   716→
   717→        {/* Gauges Row */}
   718→        {selectedDriver && (
   719→          <motion.div
   720→            key="gauges"
   721→            initial={{ opacity: 0, y: 12 }}
   722→            animate={{ opacity: 1, y: 0 }}
   723→            transition={{ duration: 0.4, delay: 0.1 }}
   724→          >
   725→            <Card className="py-4">
   726→              <CardContent>
   727→                <div className="grid grid-cols-1 sm:grid-cols-3 gap-4 items-center">
   728→                  <SpeedGauge
   729→                    speed={selectedDriver.speedKmph}
   730→                    maxSpeed={selectedDriver.maxSpeed}
   731→                  />
   732→                  <StabilityGauge score={selectedDriver.stabilityScore} />
   733→                  <EarGauge ear={selectedDriver.earRatio} />
   734→                </div>
   735→              </CardContent>
   736→            </Card>
   737→          </motion.div>
   738→        )}
   739→
   740→        {/* Metrics Grid (2x2) */}
   741→        {selectedDriver && (
   742→          <motion.div
   743→            key="metrics"
   744→            initial={{ opacity: 0, y: 12 }}
   745→            animate={{ opacity: 1, y: 0 }}
   746→            transition={{ duration: 0.4, delay: 0.2 }}
   747→            className="grid grid-cols-2 lg:grid-cols-4 gap-3"
   748→          >
   749→            <MetricCard
   750→              icon={CircleStop}
   751→              label="Harsh Braking"
   752→              value={selectedDriver.harshBrakingEvents}
   753→              unit="events"
   754→              color={selectedDriver.harshBrakingEvents > 2 ? RED : TEAL}
   755→              alert={selectedDriver.harshBrakingEvents > 2}
   756→            />
   757→            <MetricCard
   758→              icon={Zap}
   759→              label="Harsh Acceleration"
   760→              value={selectedDriver.harshAccelerationEvents}
   761→              unit="events"
   762→              color={selectedDriver.harshAccelerationEvents > 1 ? AMBER : TEAL}
   763→              alert={selectedDriver.harshAccelerationEvents > 1}
   764→            />
   765→            <MetricCard
   766→              icon={Fuel}
   767→              label="Fuel Efficiency"
   768→              value={selectedDriver.fuelEfficiency.toFixed(1)}
   769→              unit="km/L"
   770→              color={selectedDriver.fuelEfficiency > 8 ? GREEN : AMBER}
   771→            />
   772→            <MetricCard
   773→              icon={Clock}
   774→              label="Trip Duration"
   775→              value={calcTripDuration(selectedDriver.timestamp)}
   776→              unit=""
   777→              color={TEAL}
   778→            />
   779→          </motion.div>
   780→        )}
   781→
   782→        {/* Speed History + Fatigue Simulator Row */}
   783→        {selectedDriver && (
   784→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   785→            {/* Speed History Chart */}
   786→            <motion.div
   787→              key="speed-chart"
   788→              initial={{ opacity: 0, y: 12 }}
   789→              animate={{ opacity: 1, y: 0 }}
   790→              transition={{ duration: 0.4, delay: 0.3 }}
   791→            >
   792→              <Card className="py-4">
   793→                <CardHeader className="pb-0">
   794→                  <CardTitle className="text-sm flex items-center gap-2">
   795→                    <Activity className="size-4" style={{ color: TEAL }} />
   796→                    Speed History (Live)
   797→                  </CardTitle>
   798→                </CardHeader>
   799→                <CardContent>
   800→                  <SpeedChart key={speedChartKey} currentSpeed={selectedDriver.speedKmph} />
   801→                  <div className="flex items-center justify-between mt-2 text-[10px] text-muted-foreground/60">
   802→                    <span>-40s</span>
   803→                    <span className="flex items-center gap-1">
   804→                      <span className="size-1.5 rounded-full" style={{ backgroundColor: speedColor(selectedDriver.speedKmph) }} />
   805→                      {Math.round(selectedDriver.speedKmph)} km/h current
   806→                    </span>
   807→                    <span>now</span>
   808→                  </div>
   809→                </CardContent>
   810→              </Card>
   811→            </motion.div>
   812→
   813→            {/* Driver Fatigue Simulator */}
   814→            <motion.div
   815→              key="fatigue-sim"
   816→              initial={{ opacity: 0, y: 12 }}
   817→              animate={{ opacity: 1, y: 0 }}
   818→              transition={{ duration: 0.4, delay: 0.4 }}
   819→            >
   820→              <Card className="py-4">
   821→                <CardHeader className="pb-0">
   822→                  <CardTitle className="text-sm flex items-center gap-2">
   823→                    <Eye className="size-4" style={{ color: isDrowsy ? RED : TEAL }} />
   824→                    Driver Fatigue Simulator
   825→                  </CardTitle>
   826→                </CardHeader>
   827→                <CardContent>
   828→                  <EyeDiagram ear={selectedDriver.earRatio} />
   829→                  <div className="mt-2 text-[10px] text-muted-foreground/50 text-center">
   830→                    Simulated driver camera view • EAR-based eyelid tracking
   831→                  </div>
   832→                </CardContent>
   833→              </Card>
   834→            </motion.div>
   835→          </div>
   836→        )}
   837→      </div>
   838→    </ScrollArea>
   839→  );
   840→}
