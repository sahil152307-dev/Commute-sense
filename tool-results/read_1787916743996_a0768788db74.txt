    40→    40→function generatePeople(count: number, canvasW: number, canvasH: number): Person[] {
    41→    41→  const people: Person[] = [];
    42→    42→  const clothingColors = [
    43→    43→    '#e91e63', // pink/magenta
    44→    44→    '#5d4037', // dark brown

    49→    49→    '#37474f', // blue grey
    50→    50→    '#212121', // near black
    51→    51→    '#f9a825', // amber
    52→    52→    '#00695c', // teal dark
    53→    53→  ];
    54→    54→
    55→    55→  // Define zones where people can appear
    56→    56→  const zones = [
    57→    57→    // On the road / near bus
    58→    58→    { xMin: 40, xMax: canvasW * 0.45, yMin: canvasH * 0.45, yMax: canvasH * 0.75 },
    59→    59→    // Near bus shelter
    60→    60→    { xMin: canvasW * 0.55, xMax: canvasW * 0.9, yMin: canvasH * 0.35, yMax: canvasH * 0.7 },
    61→    61→    // Platform / sidewalk
    62→    62→    { xMin: 40, xMax: canvasW * 0.9, yMin: canvasH * 0.72, yMax: canvasH * 0.88 },
    63→    63→  ];
    64→    64→
    65→    65→  for (let i = 0; i < count; i++) {
    66→    66→    const zone = zones[i % zones.length];
    67→    67→    const bw = 10 + Math.random() * 6;
    68→    68→    const bh = 20 + Math.random() * 10;
    69→    69→    people.push({
    70→    70→      x: zone.xMin + Math.random() * (zone.xMax - zone.xMin),
    71→    71→      y: zone.yMin + Math.random() * (zone.yMax - zone.yMin),
    72→    72→      confidence: 0.78 + Math.random() * 0.21,
    73→    73→      id: i + 1,
    74→    74→      vx: (Math.random() - 0.5) * 0.25,
    75→    75→      vy: (Math.random() - 0.5) * 0.15,
    76→    76→      color: clothingColors[Math.floor(Math.random() * clothingColors.length)],
    77→    77→      bodyH: bh,
    78→    78→      bodyW: bw,
    79→    79→    });
    80→    80→  }
    81→    81→  return people;
    82→    82→}
